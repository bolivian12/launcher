cmake_minimum_required(VERSION 3.21)
project(EBALIA_Launcher VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network Concurrent)
find_package(LibArchive REQUIRED)
add_library(ebalia-core STATIC
  src/JavaDownloader.cpp src/LostInstaller.cpp src/PackService.cpp src/Download.cpp src/JavaRuntime.cpp src/SkinManager.cpp src/ModRepository.cpp src/Loaders.cpp src/McInstanceManager.cpp src/Archive.cpp src/JavaRunner.cpp
  src/VersionManager.cpp src/Language.cpp
  include/McInstanceManager.hpp include/JavaRunner.hpp include/VersionManager.hpp)
target_include_directories(ebalia-core PUBLIC include)
# Configure the key for this build without putting it in the source code.
set(EBALIA_CURSEFORGE_API_KEY "" CACHE STRING "CurseForge API key used by this build")
if(EBALIA_CURSEFORGE_API_KEY STREQUAL "" AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/curseforge-api-key.txt")
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/curseforge-api-key.txt" EBALIA_CURSEFORGE_API_KEY)
  string(STRIP "${EBALIA_CURSEFORGE_API_KEY}" EBALIA_CURSEFORGE_API_KEY)
endif()
configure_file(src/BuildConfig.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/generated/BuildConfig.hpp" @ONLY)
target_include_directories(ebalia-core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(ebalia-core PUBLIC Qt6::Widgets Qt6::Network Qt6::Concurrent LibArchive::LibArchive)
set(EBALIA_PATREON_SERVICE_URL "" CACHE STRING "Public HTTPS URL of the EBALIA Patreon service")
if(EBALIA_PATREON_SERVICE_URL STREQUAL "" AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/patreon-service-url.txt")
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/patreon-service-url.txt" EBALIA_PATREON_SERVICE_URL)
  string(STRIP "${EBALIA_PATREON_SERVICE_URL}" EBALIA_PATREON_SERVICE_URL)
endif()
if(NOT EBALIA_PATREON_SERVICE_URL STREQUAL "" AND NOT EBALIA_PATREON_SERVICE_URL MATCHES "^https://[A-Za-z0-9.-]+(:[0-9]+)?(/[A-Za-z0-9._~/-]*)?$")
  message(FATAL_ERROR "EBALIA_PATREON_SERVICE_URL must be a public HTTPS base URL")
endif()
configure_file(src/PatreonConfig.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/generated/PatreonConfig.hpp" @ONLY)
qt_add_executable(ebalia-launcher src/main.cpp src/MainWindow.cpp src/CreatorDialog.cpp src/HomeBanner.cpp include/HomeBanner.hpp src/SkinsPage.cpp src/SetupDialog.cpp src/PackBrowser.cpp src/CreateInstanceDialog.cpp src/InstanceViews.cpp src/InstanceIcons.cpp src/Icons.cpp src/Ui.cpp src/CommunityPage.cpp src/PatreonAuth.cpp src/PatreonNewsPage.cpp src/UpdateChecker.cpp include/UpdateChecker.hpp src/AccountManager.cpp src/MsAuth.cpp
  include/MainWindow.hpp include/Ui.hpp include/PatreonAuth.hpp include/AccountManager.hpp include/MsAuth.hpp
  resources/launcher.qrc)
target_link_libraries(ebalia-launcher PRIVATE ebalia-core)
target_compile_definitions(ebalia-launcher PRIVATE EBALIA_APP_VERSION="${PROJECT_VERSION}")
target_include_directories(ebalia-launcher PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(WIN32)
  set_target_properties(ebalia-launcher PROPERTIES WIN32_EXECUTABLE TRUE)
  target_sources(ebalia-launcher PRIVATE resources/ebalia.rc)
  target_include_directories(ebalia-launcher PRIVATE resources)
  if(MSVC)
    target_link_options(ebalia-launcher PRIVATE /MANIFEST:NO)
    target_compile_options(ebalia-core PUBLIC /utf-8)
  endif()
elseif(APPLE)
  target_sources(ebalia-launcher PRIVATE resources/ebalia.icns)
  set_source_files_properties(resources/ebalia.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  set_target_properties(ebalia-launcher PROPERTIES MACOSX_BUNDLE TRUE
    MACOSX_BUNDLE_ICON_FILE "ebalia.icns"
    MACOSX_BUNDLE_BUNDLE_NAME "EBALIA Launcher" MACOSX_BUNDLE_GUI_IDENTIFIER "com.ebalia.launcher"
    MACOSX_BUNDLE_BUNDLE_VERSION "1.0.0" MACOSX_BUNDLE_SHORT_VERSION_STRING "1.0.0")
endif()
include(GNUInstallDirs)
install(TARGETS ebalia-launcher BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(NOT APPLE AND NOT WIN32)
  install(FILES packaging/ebalia-launcher.desktop DESTINATION share/applications)
  foreach(icon_size 16 32 48 64 128 256 512)
    install(FILES resources/icons/app/${icon_size}.png DESTINATION share/icons/hicolor/${icon_size}x${icon_size}/apps RENAME ebalia-launcher.png)
  endforeach()
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME patreon-service COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/services/patreon -p test_server.py)
    add_test(NAME platform-icons COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/packaging/verify-icons.py --source ${CMAKE_CURRENT_SOURCE_DIR})
  endif()
  find_package(Qt6 REQUIRED COMPONENTS Test)
  qt_add_executable(ebalia-tests tests/CoreTests.cpp resources/launcher.qrc)
  target_link_libraries(ebalia-tests PRIVATE ebalia-core Qt6::Test)
  qt_add_executable(ebalia-ui-tests tests/UiTests.cpp src/MainWindow.cpp src/CreatorDialog.cpp src/HomeBanner.cpp include/HomeBanner.hpp src/SkinsPage.cpp src/SetupDialog.cpp src/PackBrowser.cpp src/CreateInstanceDialog.cpp src/InstanceViews.cpp src/InstanceIcons.cpp src/Icons.cpp src/Ui.cpp src/CommunityPage.cpp src/PatreonAuth.cpp src/PatreonNewsPage.cpp src/UpdateChecker.cpp include/UpdateChecker.hpp src/AccountManager.cpp src/MsAuth.cpp
      include/MainWindow.hpp include/Ui.hpp include/PatreonAuth.hpp include/AccountManager.hpp include/MsAuth.hpp
    resources/launcher.qrc)
  set_target_properties(ebalia-ui-tests PROPERTIES WIN32_EXECUTABLE FALSE)
  target_link_libraries(ebalia-ui-tests PRIVATE ebalia-core Qt6::Test)
  target_include_directories(ebalia-ui-tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  add_test(NAME ui COMMAND ebalia-ui-tests -o ui-test-results.txt,txt -o -,txt)
  if(WIN32)
    set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_FONTDIR=$ENV{SystemRoot}/Fonts")
  else()
    set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endif()
  add_test(NAME core COMMAND ebalia-tests)
  set_tests_properties(core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
set(CPACK_PACKAGE_VENDOR "EBALIA")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
include(CPack)
